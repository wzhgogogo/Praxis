import assert from 'node:assert/strict';
import {readFile} from 'node:fs/promises';
import test from 'node:test';
import {BrowserTaskExecutor} from './browser-task-executor.js';
import {decodeBrowserReadActionWire} from './browser-action-decision.js';
import type {BrowserPageControl,BrowserSession} from './browser-runtime.js';
const data=JSON.parse(await readFile(new URL('./fixtures/advance-authoritative/r1.json',import.meta.url),'utf8'));
// Actual recorded observations, not a complete website replay. Group keys and
// native tag are local reconstruction of the recorded group refs/type=button.
function controls(frame:any):BrowserPageControl[]{return frame.observation.targets.map((t:any)=>({...t,id:t.ref,stableKey:[t.kind,t.label,t.value??'',t.controlGroupRef?.split(':group:')[1]].join('|'),controlGroupKey:'recorded-group:'+t.controlGroupRef.split(':group:')[1],visible:true,structure:{tag:t.kind==='BUTTON'?'BUTTON':t.kind==='LINK'?'A':'INPUT',name:'',classes:[],dialogLabel:'',formClass:'',sliderCount:0}}));}
for(const [name,indices,field] of [['people',[2,3,4],'PARTY_SIZE'],['calendar',[0,1],'DATE']] as const)test(`D02-V r1 ${name}: one decision advances until authoritative value appears`,async t=>{
 const frames=indices.map(i=>data.frames[i]);let index=0,calls=0,clicks=0;const events:any[]=[];
 const original=frames[0].originalAction,chosen=frames[0].observation.targets.find((x:any)=>x.ref===original.targetRef);assert.ok(chosen);
 const session={metadata:{runtimeProvider:'LOCAL_PLAYWRIGHT_CHROMIUM',engine:'CHROMIUM',startedAt:'2026-11-12T00:00:00Z',readNetworkBoundary:'INSTALLED'},async snapshot(){return {url:frames[index].observation.url,title:'Recorded source',text:'Recorded widget',html:'',settlement:'SETTLED'};},async observeControls(){return controls(frames[index]);},async click(id:string){assert.equal(controls(frames[index]).find(x=>x.id===id)?.label,chosen.label);index++;clicks++;},async waitForSettled(){return true;},async close(){}} as unknown as BrowserSession;
 const executor=new BrowserTaskExecutor({openSession:async()=>session},{maxAutomaticElapsedMs:20000,maxOperationsPerCandidate:50,maxModelCallsPerCandidate:4,maxModelCallsTotal:4,onDiagnostic:e=>events.push(e),modelDecision:{async decide(input){calls++;if(calls===1){const target=input.observation.targets.find(t=>t.label===chosen.label&&t.controlGroupLabel===chosen.controlGroupLabel);assert.ok(target);return {type:'ADVANCE_TO_AUTHORITATIVE',targetRef:target.ref,field,reason:'Recorded same-group advancement'} as any;}
 assert.equal(index,frames.length-1,'hand back only after requested value appears');assert.ok(input.observation.targets.some(t=>t.availableActions?.includes('CLICK_AUTHORITATIVE:'+field)),'existing authoritative choice remains available');return {type:'REQUEST_HUMAN_HELP',reason:'End this component at the hand-back; no selection or availability claim'};}}});
 try{const result=await executor.runSkill({taskId:'recorded-widget',source:'recorded',stage:'AVAILABILITY',session,signal:new AbortController().signal,allowedOrigins:[new URL(frames[0].observation.url).origin],goal:frames[0].goal,objective:'Advance model-selected observed group',completion:()=>({complete:false,reason:'This component never establishes inventory'})});assert.equal(result.status,'REQUESTED_HUMAN_HELP');assert.equal(clicks,frames.length-1);assert.equal(calls,2,'no intermediate model decisions');assert.ok(events.some(e=>e.detail==='ADVANCE_TARGET_FOUND'));
 const originalAdvanceCalls=frames.length-1,avoided=originalAdvanceCalls-1,projectedSavedMs=frames.slice(1,-1).reduce((sum:number,f:any)=>sum+f.latencyMs,0);t.diagnostic(JSON.stringify({mode:'RECORDED_COMPONENT_ONLY',originalAdvanceCalls,newAdvanceCalls:1,avoidedModelCalls:avoided,projectedSavedMs,note:'Reuses first recorded model latency; no new-model latency or Live success asserted'}));}finally{await executor.closeAll();}
});
test('new advance action is accepted on the strict wire',()=>{assert.equal(decodeBrowserReadActionWire({action:'ADVANCE_TO_AUTHORITATIVE',targetRef:'observation:1:target:1',authoritativeField:'TIME',requestedState:'NONE',reason:'bounded advancement'},new Set(['observation:1:target:1'])).type,'ADVANCE_TO_AUTHORITATIVE');});
